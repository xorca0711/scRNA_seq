name: Repository checks

on:
  push:
  pull_request:

permissions:
  contents: read

jobs:
  validate:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - name: Compile Python sources
        run: python -m compileall -q analysis Thesis
      - name: Test evidence and provenance contracts
        run: python -m unittest discover -s analysis/tests -q
      - name: Check generated claim manifest and summaries
        run: python analysis/scripts/claim_contract.py --check
      - name: Verify tracked Nb1 evidence
        run: python Thesis/gate1_03_nabhan_2018/nb1/verify_outputs.py
      - name: Validate tracked artefacts and documentation
        run: python analysis/scripts/validate_repository.py
